Cleaning data

# Making categorical variables
gtfs <- gtfs %>%
  mutate(
    agency = factor(agency, levels = c("MBTA", "MTA", "WMATA")),
    direction = factor(direction),
    route_id = factor(route_id),
    day_of_week = factor(
      day_of_week,
      levels = c("0", "1", "2", "3", "4", "5", "6"),
      labels = c("Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday")
    ),
    day_of_week = relevel(day_of_week, ref = "Friday")
  )

Regressions

# Model 1: Pooled OLS, no fixed effects
model1 <- plm(
  deviation ~ agency + scheduled_headway + peak + direction +
    n_stops + timepoint_rate + route_length_mi + headway_cv,
  data = gtfs,
  model = "pooling"
)

# Model 2: Pooled OLS, day fixed effects
model2 <- plm(
  deviation ~ agency + scheduled_headway + peak + direction +
    n_stops + timepoint_rate + route_length_mi + headway_cv +
    day_of_week,
  data = gtfs,
  model = "pooling"
)

# Model 6: Logistic regression with bunching as outcome
model6 <- glm(
  bunching ~ agency + scheduled_headway + peak + direction +
    n_stops + timepoint_rate + route_length_mi + headway_cv,
  data = gtfs,
  family = "binomial"
)

summary(model6)

marginal_effects <- margins(model6) # getting the marginl effects
